public|apply_successful_payment|CREATE OR REPLACE FUNCTION public.apply_successful_payment()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    IF NEW.status = 'successful' AND OLD.status IS DISTINCT FROM 'successful' THEN
        NEW.paid_at := COALESCE(NEW.paid_at, now());

        IF NEW.purpose = 'ticket' THEN
            UPDATE tickets
               SET status = 'confirmed'
             WHERE payment_id = NEW.id AND status = 'pending';

        ELSIF NEW.purpose = 'membership_dues' THEN
            UPDATE memberships m
               SET status     = 'active',
                   start_date = CURRENT_DATE,
                   end_date   = CURRENT_DATE + p.duration_days
              FROM membership_plans p
             WHERE m.id = NEW.membership_id
               AND p.id = m.plan_id
               AND m.status = 'pending';
        END IF;
    END IF;
    RETURN NEW;
END;
$function$

public|check_ticket_type_event|CREATE OR REPLACE FUNCTION public.check_ticket_type_event()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM ticket_types
        WHERE id = NEW.ticket_type_id AND event_id = NEW.event_id
    ) THEN
        RAISE EXCEPTION 'Ticket type % does not belong to event %',
            NEW.ticket_type_id, NEW.event_id;
    END IF;
    RETURN NEW;
END;
$function$

public|enforce_event_capacity|CREATE OR REPLACE FUNCTION public.enforce_event_capacity()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_capacity INT;
    v_taken    INT;
BEGIN
    IF NEW.status = 'cancelled' THEN
        RETURN NEW;
    END IF;

    SELECT capacity INTO v_capacity
    FROM events WHERE id = NEW.event_id FOR UPDATE;

    SELECT COUNT(*) INTO v_taken
    FROM tickets
    WHERE event_id = NEW.event_id
      AND status <> 'cancelled'
      AND id IS DISTINCT FROM NEW.id;

    IF v_taken >= v_capacity THEN
        RAISE EXCEPTION 'Event % is sold out (capacity %)', NEW.event_id, v_capacity
            USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$function$
